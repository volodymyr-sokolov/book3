\chapter*{Перелік умовних скорочень}
\addcontentsline{toc}{chapter}{Перелік умовних скорочень}
\markboth{Перелік умовних скорочень}{Перелік умовних скорочень}

\begin{description}[leftmargin=!,labelwidth=22mm,font=\normalfont\sffamily,itemsep=1pt]
\item[АСК ТП] автоматизована система керування технологічними процесами
\item[ЗПК] злоякісний програмний код
\item[КК] Кримінальний кодекс України
\item[МН] машинне навчання
\item[ОС] операційна система
\item[ПЗ] програмне забезпечення
\item[AMSI] Antimalware Scan Interface
\item[APT] Advanced Persistent Threat (цілеспрямована стала загроза)
\item[ATT\&CK] Adversarial Tactics, Techniques, and Common Knowledge (MITRE)
\item[C2] Command and Control (командний сервер керування)
\item[CERT-UA] Урядова команда реагування на комп'ютерні надзвичайні події України
\item[CFG] Control-Flow Graph (граф потоку керування)
\item[CI/CD] Continuous Integration / Continuous Delivery
\item[DDoS] Distributed Denial of Service
\item[EDR] Endpoint Detection and Response
\item[ELF] Executable and Linkable Format
\item[ETW] Event Tracing for Windows
\item[ICS] Industrial Control System (промислова система керування)
\item[IoC] Indicator of Compromise (індикатор компрометації)
\item[IoT] Internet of Things (Інтернет речей)
\item[IT/OT] Information Technology / Operational Technology
\item[LotL] Living off the Land
\item[PE] Portable Executable
\item[PPV] Positive Predictive Value (прогностична цінність)
\item[ROC] Receiver Operating Characteristic
\item[SBOM] Software Bill of Materials
\item[SCADA] Supervisory Control and Data Acquisition
\item[SIEM] Security Information and Event Management
\item[SLSA] Supply-chain Levels for Software Artifacts
\item[SOC] Security Operations Center (центр безпеки)
\item[STIX] Structured Threat Information Expression
\item[TAXII] Trusted Automated Exchange of Intelligence Information
\item[TTP] Tactics, Techniques, and Procedures
\item[XDR] Extended Detection and Response
\item[YARA] Yet Another Recursive Acronym (мова правил виявлення)
\end{description}
